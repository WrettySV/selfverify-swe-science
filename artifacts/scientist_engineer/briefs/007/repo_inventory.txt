def _load_module(name: str, path: Path) -> ModuleType:
    spec = importlib.util.spec_from_file_location(name, path)
    if spec is None or spec.loader is None:
        raise ImportError(f"Cannot load {name} from {path}")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

def residual_ratio(series: list[np.ndarray], reference: np.ndarray) -> float:
    denom = np.linalg.norm(reference)
    return float(
        np.mean(
            [np.linalg.norm(np.asarray(item) - reference) / denom for item in series]
        )
    )


def average_snr(series: list[np.ndarray]) -> float:
    spectrum = np.fft.fftshift(np.fft.fft(np.mean(np.asarray(series), axis=0)))
    magnitude = np.abs(spectrum)
    edge = max(8, len(magnitude) // 12)
    noise = np.concatenate([magnitude[:edge], magnitude[-edge:]])
    return float(np.max(magnitude) / (np.std(noise) + 1e-12))


def linewidth_hz(series: list[np.ndarray], dt: float) -> float:
    spectrum = np.fft.fftshift(np.fft.fft(np.mean(np.asarray(series), axis=0)))
    magnitude = np.abs(spectrum)
    peak = int(np.argmax(magnitude))
    half_height = magnitude[peak] / 2.0
    left = peak
    while left > 0 and magnitude[left] >= half_height:
        left -= 1
    right = peak
    while right < len(magnitude) - 1 and magnitude[right] >= half_height:
        right += 1
    return float((right - left) * (1.0 / dt) / len(magnitude))

The script loads `fixtures/public_dynamic_series.npz`, runs the package's
dynamic spectral-registration workflow, compensates the estimated drift, and
reports before/after quality metrics in `outputs/reproduction_report.json`.
The workflow uses a stable registration window, records how many transients are
accepted for averaging, and records residual, SNR, and linewidth observations
before and after correction.

`workflow_completed` means that the candidate workflow returned a finite,
inspectable registered series. The public report does not apply fixed
improvement multipliers or certify the repair. Private verification owns the
authoritative scientific magnitudes and additional cases. If no transient is
accepted, or if the workflow returns a non-finite scientific observation, the
script emits a structured `pre_fix_expected_failure` report with exit code 1.
Import, fixture, dependency, and result-protocol exceptions are reported as
`runner_failure` with exit code 2. All reports are strict JSON; unavailable
observations are represented as `null`, never `NaN` or `Infinity`.
### paper.md
# Spectral Registration Context For Dynamic MRS

Magnetic-resonance spectroscopy experiments often acquire many repeated
transients before averaging them. Small scanner and subject changes can shift
individual transients in frequency and phase. If those shifts are not corrected,
averaging broadens peaks, distorts line shapes, and reduces the effective SNR.

Near et al. introduced time-domain spectral registration as a practical way to
estimate frequency and phase drift between an input transient and a reference.
The method adjusts a transient with a complex exponential and chooses the
frequency/phase parameters that minimize the time-domain mismatch to the
reference. After a valid correction, the aligned transients should be more
similar to each other, and their averaged spectrum should have a sharper,
higher-quality peak than the uncorrected average.

For this task, treat the public data as a compact dynamic MRS quality-control
case. The important scientific invariant is qualitative but testable:

```text
applying the registration workflow should improve alignment quality.
```

In practice, not every spectral feature is equally useful for registration.
Some peaks can change with experimental condition, editing state, or nuisance
contamination, while another spectral region remains a stable reference. A
registration workflow should therefore be able to compare transients in the
scientifically appropriate frequency window, then apply the resulting
time-domain correction to the full transient.

Useful checks include reduced residual mismatch to the reference in the stable
region, narrower dominant spectral peaks after averaging, improved SNR
proxies, and avoiding contamination from transients that cannot be registered
to the common reference. A repair that only works for one point count, one
fixture, or one globally identical spectrum is not scientifically adequate.

def spectral_registration(data, target, initial_guess=(0.0, 0.0), frequency_range=None):
    """
    Performs the spectral registration method to calculate the frequency and
    phase shifts between the input data and the reference spectrum target. The
    frequency range over which the two spectra are compared can be specified to
    exclude regions where the spectra differ.

    Parameters
    ----------
    data : MRSData
    target : MRSData
    initial_guess : tuple
    frequency_range : 

    Returns
    -------

    """

    # make sure that there are no extra dimensions in the data
    data = data.squeeze()
    target = target.squeeze()

    # the supplied frequency range can be none, in which case we use the whole
    # spectrum, or it can be a tuple defining two frequencies in Hz, in which
    # case we use the spectral points between those two frequencies, or it can
    # be a numpy.array of the same size as the data in which case we simply use
    # that array as the weightings for the comparison
    if type(frequency_range) is tuple:
        spectral_weights = frequency_range[0] < data.frequency_axis() & data.frequency_axis() < frequency_range[1]
    else:
        spectral_weights = frequency_range

    # define a residual function for the optimizer to use
    def residual(input_vector):
        transformed_data = transform_fid(data, input_vector[0], input_vector[1])
        residual_data = transformed_data - target
        if frequency_range is not None:
            spectrum = residual_data.spectrum()
            weighted_spectrum = residual_data * spectral_weights
            # remove zero-elements
            weighted_spectrum = weighted_spectrum[weighted_spectrum != 0]
            residual_data = numpy.fft.ifft(numpy.fft.ifftshift(weighted_spectrum))
        return_vector = numpy.zeros(len(residual_data) * 2)
        return_vector[:len(residual_data)] = residual_data.real
        return_vector[len(residual_data):] = residual_data.imag
        return return_vector

    return -out[0][0], -out[0][1]

def register_dynamic_series(
    series,
    reference=None,
    initial_guesses=None,
    frequency_range=None,
    reject_threshold=None,
):
    """
    Register a dynamic MRS series to a common reference.

    The returned dictionary is intentionally simple so that downstream quality
    control can inspect the corrected transients, per-transient drift estimates,
    residuals, and accepted mask.
    """
    transients = [item.squeeze() for item in series]
    if len(transients) == 0:
        raise ValueError("series must contain at least one transient")
    if reference is None:
        reference = dynamic_reference(transients)
    else:
        reference = reference.squeeze()

    if initial_guesses is None:
        initial_guesses = [(0.0, 0.0)] * len(transients)
    if len(initial_guesses) != len(transients):
        raise ValueError("initial_guesses must match the number of transients")

    corrected = []
    estimates = []
    before = []
    after = []
    accepted = []

    for transient, initial_guess in zip(transients, initial_guesses):
        before.append(residual_ratio(transient, reference, frequency_range=None))
        freq, phase = spectral_registration(
            transient,
            reference,
            initial_guess=initial_guess,
            frequency_range=None,
        )
        aligned = transform_fid(transient, -freq, -phase)
        corrected.append(aligned)
        estimates.append((freq, phase))
        after_value = residual_ratio(aligned, reference, frequency_range=None)
        after.append(after_value)
        accepted.append(True)

    return {
        "reference": reference,
        "corrected": corrected,
        "estimates": numpy.asarray(estimates, dtype=float),
        "residuals_before": numpy.asarray(before, dtype=float),
        "residuals_after": numpy.asarray(after, dtype=float),
        "accepted": numpy.asarray(accepted, dtype=bool),
        "frequency_range": frequency_range,
    }

        """

        See Also
