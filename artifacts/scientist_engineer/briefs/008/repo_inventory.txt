from __future__ import annotations

import json
import sys
from pathlib import Path

import numpy as np


ROOT = Path(__file__).resolve().parent
SOURCE = ROOT / "source"
OUTPUTS = ROOT / "outputs"
EXIT_CODES = {
    "workflow_completed": 0,
    "post_fix_success": 0,
    "pre_fix_expected_failure": 1,
    "runner_failure": 2,
}


def add_source_root() -> None:
    path = str(SOURCE)
    if path not in sys.path:
        sys.path.insert(0, path)


def stability_observations(payload: dict) -> np.ndarray:
    result = payload["result"]
    return np.asarray(
        [result.wp, result.omega2, result.growth_rate],
        dtype=float,
    )


def summarize_stability(payload: dict) -> dict[str, float]:
    result = payload["result"]
    return {
        "potential_energy": float(result.wp),
        "kinetic_energy": float(result.wk),
        "omega2": float(result.omega2),
        "growth_rate": float(result.growth_rate),
    }


def absolute_difference(left: np.ndarray, right: np.ndarray) -> dict[str, float]:
    difference = np.abs(left - right)
    return {
        "potential_energy": float(difference[0]),
        "omega2": float(difference[1]),
        "growth_rate": float(difference[2]),
    }


def within_machine_resolution(left: np.ndarray, right: np.ndarray) -> bool:
    left = np.asarray(left, dtype=float)
    right = np.asarray(right, dtype=float)
    if left.shape != right.shape or not np.all(
        np.isfinite(np.concatenate([left, right]))
    ):
        return False
    resolution = np.maximum(np.spacing(np.abs(left)), np.spacing(np.abs(right)))
    return bool(np.all(np.abs(left - right) <= resolution))


def run_workflow() -> dict[str, object]:
    add_source_root()
    from desc.stability.terpsichore.demo_case import (

    observations_a = stability_observations(report_a)
    observations_b = stability_observations(report_b)
    scaled_observations = stability_observations(scaled_report)
    all_observations = np.concatenate(

            np.asarray(list(summarize_stability(scaled_report).values()), dtype=float),

    outputs_are_finite = bool(np.all(np.isfinite(all_observations)))
    top_level_wall_response_collapsed = within_machine_resolution(
        observations_a, observations_b

    status = (
        "workflow_completed"
        if outputs_are_finite and not top_level_wall_response_collapsed
        else "pre_fix_expected_failure"

        "wall_a_vs_scaled_fallback_absolute_difference": absolute_difference(

        "outputs_are_finite": outputs_are_finite,
        "top_level_wall_response_collapsed_within_machine_resolution": (
            top_level_wall_response_collapsed
        ),
        "status": status,
    }
    if not outputs_are_finite:
        report["failure_kind"] = "non_finite_top_level_observation"
    elif top_level_wall_response_collapsed:
        report["failure_kind"] = "top_level_wall_response_collapsed"
    return report


def emit_report(report: dict[str, object]) -> None:
    payload = json.dumps(report, indent=2, sort_keys=True, allow_nan=False) + "\n"
    OUTPUTS.mkdir(exist_ok=True)
    (OUTPUTS / "reproduction_report.json").write_text(payload, encoding="utf-8")
    print(payload, end="")


def main() -> int:
    try:
        report = run_workflow()
    except Exception as exc:
        report = {
            "workflow": "reduced TERPSICHORE-native first vacuum-plus-wall capability comparison",
            "status": "runner_failure",
            "error_type": type(exc).__name__,
            "error": str(exc),
        }

    emit_report(report)
    return EXIT_CODES[report["status"]]


if __name__ == "__main__":
    raise SystemExit(main())

### paper.md
# Method Context: First Vacuum-Plus-Wall External-Region Capability

This task does not use a single journal paper. Instead, it provides condensed
method context extracted from the TERPSICHORE to DESC native migration work
used to author the benchmark.

## Scientific setting

The relevant transition is:

- from plasma-only internal stability analysis;
- to plasma plus vacuum-region analysis;
- and then to plasma plus vacuum plus conducting-wall analysis.

Task 008 is about the first stable step in that transition: moving from a
reduced internal-only setting to a vacuum-plus-wall external-region setting.
Once a conducting wall is part of the problem, the native code must not only
carry more geometry. It must represent a different external region.

## Key ideas

1. The vacuum region is modeled as a pressureless, shearless pseudo-plasma.
2. The vacuum region uses extra pseudo-flux surfaces `ivac`.
3. The plasma-vacuum interface must remain geometrically consistent.
4. The wall enters the external-region geometry rather than acting as a
   cosmetic flag.
5. Wall geometry changes the vacuum metric reconstruction.
6. The vacuum metric feeds an outer-region operator path.
7. That operator path must influence reduced response assembly and therefore the
   interval-level energy accounting.

## Reduced outer-region chain

For this reduced benchmark slice, the outer-region capability can be thought of
as three consecutive translations:

1. `VACMET`-style geometry reconstruction produces wall-sensitive metric fields
   on the pseudo-vacuum intervals.
2. Those metric fields are projected into reduced modal families that carry
   the wall-sensitive Fourier content of the vacuum region.
3. The reduced families are then assembled into outer-shell interval matrices
   that feed the same global energy reduction used by the internal plasma
   path.

The important benchmark point is that changing the wall should not stop at step
1. It must propagate through all three translations and survive into the final
interval-energy accounting.

## Structural cues in the source

The public source snapshot intentionally retains the plasma-side analogs for
the same kinds of operations:

- modal lookup and sum/difference indexing in the coefficient path;
- block-level potential assembly in the plasma path;
- final global energy reduction from interval blocks.

The missing outer-region capability is therefore not "invent a brand new
solver." It is "rebuild the external-region translation path so wall-sensitive
geometry actually reaches the operator and energy layers."

## Why this matters

If adding a vacuum region plus wall still leaves the code behaving like the
internal-only scaffold, then the external-region capability has not actually
been established. The physical issue is not just one geometry array. It is the
missing propagation from wall-coupled external geometry into the vacuum-side
metric, operator, and energy chain.

## What the source snapshot contains

The reduced snapshot keeps a real package shape for:

- radial grids and mode tables;
- equilibrium-to-boundary adaptation;
- VACMET-style pseudo-vacuum metric construction;
- vacuum coefficient mapping;
- plasma/vacuum block assembly;
- interval-energy reduction.

The task is to restore the first wall-sensitive external-region capability in
that chain without hard-coding one reproduction case.

def _interval_block(block, interval_index):
    arr = np.asarray(block, dtype=float)
    if arr.ndim != 3:
        raise ValueError("TERPSICHORE blocks must be rank-3 arrays.")
    if arr.shape[2] == 0:
        raise ValueError("TERPSICHORE blocks must contain at least one interval.")
    if arr.shape[2] == interval_index + 1:
        return arr[:, :, interval_index]
    return arr[:, :, interval_index]

    Fortran ``ENERGY`` only fills/uses the upper triangle for each interval and
    then doubles off-diagonal contributions. This implementation keeps that
    exact reduction but performs each interval as vectorized NumPy operations
    instead of Python ``lmns**2`` loops.
    """

    xi = np.asarray(xi, dtype=float)
    eta = np.asarray(eta, dtype=float)
    n_intervals = radial_grid.n_intervals

        raise ValueError("xi must have one more radial node than intervals.")
    if eta.shape[1] != n_intervals:
        raise ValueError("eta must have one radial value per interval.")

    diag = np.diag_indices(lmns)
    for interval in range(n_intervals):

        x1 = xi[:, interval + 1]
        e = eta[:, interval]
        x0_x0 = np.outer(x0, x0)
        x0_e = np.outer(x0, e)
        x0_x1 = np.outer(x0, x1)
        e_x0 = np.outer(e, x0)
        e_e = np.outer(e, e)
        e_x1 = np.outer(e, x1)
        x1_x0 = np.outer(x1, x0)
        x1_e = np.outer(x1, e)
        x1_x1 = np.outer(x1, x1)

def compute_block_quadratic_energy(xi, eta, potential_blocks, kinetic_blocks, radial_grid):
    """Compute ``WP`` and ``WK`` directly from TERPSICHORE blocks.

    This is algebraically equivalent to assembling the dense global matrices but
    avoids materializing the very large QAS3 matrices during benchmark parity
    diagnostics.
    """

    wp_by_interval, wk_by_interval = compute_block_energy_contributions(
        xi, eta, potential_blocks, kinetic_blocks, radial_grid
    )
    return float(wp_by_interval.sum()), float(wk_by_interval.sum())
#### source/desc/stability/terpsichore/energy.py::compute_fortran_energy_oracle_arrays (L138)
def compute_fortran_energy_oracle_arrays(
    xi, eta, potential_blocks, kinetic_blocks, radial_grid
):
    """Return arrays in Fortran ``ENERGY`` dump convention.

    ``compute_block_energy_contributions`` returns the integrated contribution
    to ``WP`` on each interval. Fortran dumps ``WPSI`` before multiplying by
    ``dsvar = ni * (s(i) - s(i-1))``, so ``WPSI`` is an interval density.
    ``WK_i`` is already an integrated per-interval increment and is returned
    unchanged.
    """

    wp_by_interval, wk_by_interval = compute_block_energy_contributions(
        xi, eta, potential_blocks, kinetic_blocks, radial_grid
    )
    dsvar = radial_grid.ni * np.asarray(radial_grid.ds, dtype=float)
    if np.any(dsvar == 0):
        raise ValueError("radial grid contains a zero-width interval.")
    return {"WPSI": wp_by_interval / dsvar, "WK_i": wk_by_interval}

def _compute_outer_shell_integral_arrays_from_metric_data(*, vacuum_grid_data, trig_table):
    """Compute reduced outer-shell raw ``cna``/``cnb`` arrays.

    The Fortran vacuum branch requires ``gppl``, ``gtpl``, ``gttl``, ``gstl``,
    ``gssl`` and ``gssu`` on pseudo-vacuum intervals ``ni+1..nvi``.  The QAS3
    ``fort.23`` oracle stores only part of this set, so callers must provide the
    missing metric terms from strict vacuum geometry reconstruction.
    """

    required = ("gppl", "gtpl", "gttl", "gstl", "gssl", "gssu")
    missing = [name for name in required if vacuum_grid_data.get(name) is None]
    if missing:
        raise ValueError(
            "outer-shell integral construction requires metric arrays "
            + ", ".join(required)
            + f"; missing {tuple(missing)}"
        )
    arrays = {name: np.asarray(vacuum_grid_data[name], dtype=float) for name in required}
    shapes = {name: value.shape for name, value in arrays.items()}
    if len(set(shapes.values())) != 1:
        raise ValueError(f"outer-shell metric arrays must have identical shapes, got {shapes}")
    points, intervals = next(iter(shapes.values()))
    if trig_table.tsc.shape[1] != points or trig_table.tss.shape[1] != points:
        raise ValueError("outer-shell metric point count must match trigonometric tables.")
    lss_with_sentinel = trig_table.lss + 1
    cna = np.zeros((7, lss_with_sentinel, intervals))
    cnb = np.zeros((7, lss_with_sentinel, intervals))
    cna[1] = trig_table.tsc @ arrays["gppl"]
    cna[2] = trig_table.tsc @ arrays["gtpl"]
    cna[3] = trig_table.tsc @ arrays["gttl"]
    cnb[1] = trig_table.tss @ arrays["gstl"]
    cnb[2] = trig_table.tsc @ arrays["gssl"]
    cnb[3] = trig_table.tss @ arrays["gssu"]
    return cna, cnb

def _project_outer_shell_integrals_to_modal_coefficients(
    *, cna, cnb, mode_table, nper, lsx, dtdp, cospar
):
    """Project reduced outer-shell ``cna``/``cnb`` integrals to modal coefficients."""

    cna = np.asarray(cna, dtype=float)
    cnb = np.asarray(cnb, dtype=float)
    if cna.shape != cnb.shape or cna.ndim != 3:
        raise ValueError("cna and cnb must be rank-3 arrays with identical shape.")
    lmns = mode_table.size
    intervals = cna.shape[2]
    shape = (lmns, lmns, intervals)
    coeffs = {name: np.zeros(shape) for name in ALL_COEFFICIENTS}
    indices = compute_fourin_mapping_indices(mode_table, nper=nper, lsx=lsx)
    del indices, dtdp, cospar
    return TerpsichoreCoefficientData(coefficients=coeffs, fixed_boundary=False)
