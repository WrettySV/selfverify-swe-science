from __future__ import annotations

import json
import os
import sys
from pathlib import Path
from typing import Any


TASK_ROOT = Path(__file__).resolve().parent
SOURCE_ROOT = TASK_ROOT / "source"
REPORT = TASK_ROOT / "outputs" / "public_reproduction_report.json"
FIXTURE = TASK_ROOT / "fixtures" / "magnetic_setting_records.json"


def _write_report(payload: dict[str, Any]) -> int:
    REPORT.parent.mkdir(exist_ok=True)
    text = json.dumps(payload, indent=2, sort_keys=True) + "\n"
    REPORT.write_text(text, encoding="utf-8")
    print(text, end="")
    return {"workflow_completed": 0, "runner_failure": 2}[payload["status"]]


def main() -> int:
    os.environ.setdefault("PYTHONHASHSEED", "0")
    sys.path.insert(0, str(SOURCE_ROOT))
    sys.path.insert(0, str(TASK_ROOT))

    try:
        from workflow.magnetic_catalogue import catalogue_report, load_records

        report = catalogue_report(load_records(FIXTURE))
        return _write_report(
            {
                "schema_version": "1.0",
                "status": "workflow_completed",
                "observations": {
                    "record_count": report["record_count"],
                    "resolved_count": report["resolved_count"],
                    "failure_count": len(report["failures"]),
                    "operation_counts": [
                        item["operation_count"] for item in report["batch_observations"]
                    ],
                    "finite_summaries": len(report["batch_observations"]),
                },
            }
        )
    except Exception as exc:
        return _write_report(
            {
                "schema_version": "1.0",
                "status": "runner_failure",
                "error_type": type(exc).__name__,
                "error": str(exc)[:240],
            }
        )


if __name__ == "__main__":
    raise SystemExit(main())

# Public Reproduction

The public reproducer loads a magnetic archive record and asks the retained
source snapshot to produce the magnetic operation family used by downstream
catalogue analysis.

Expected pre-fix behavior:

```bash
python reproduce.py
```

returns exit code `0` and writes `outputs/public_reproduction_report.json` with
finite archive observations.

The public status is diagnostic only.  It does not decide the complete

### paper.md
# Scientific Context

This task uses a retained subset of pymatgen, an open-source materials-analysis
library described in:

- Shyue Ping Ong et al., "Python Materials Genomics (pymatgen): A robust,
  open-source python library for materials analysis", Computational Materials
  Science 68, 314-319, 2013. DOI: 10.1016/j.commatsci.2012.10.028.

Magnetic space groups extend crystallographic symmetry operations with
time-reversal labels used to describe magnetic structures.  Materials archives
may store records in different coordinate conventions.  A scientific workflow
that reconstructs operation families from archive records should preserve the
meaning of valid producer records through downstream analysis.

The retained source includes the project code and magnetic-space-group database
needed for the offline reproduction.

def _catalogue_coordinates(xyz: list[float], setting: str) -> list[float]:
    matrix, origin = _setting_matrix(setting)
    return [round(float(value), 8) for value in np.mod(matrix @ np.array(xyz) + origin, 1.0)]

def operation_family(record: MagneticRecord) -> dict[str, object]:
    from pymatgen.io.cif import CifParser, MagneticCifBlock

    block = MagneticCifBlock(
        {
            "_space_group_magn.number_BNS": ".".join(str(part) for part in record.bns_number),
            "_space_group_magn.transform_BNS_Pp_abc": record.archive_basis,
        }
    )
    ops = CifParser().get_magsymops(block)
    if isinstance(ops, Exception):
        raise ops
    transformed = []
    for op in ops:
        xyz = np.mod(op.operate(record.probe_fractional_site), 1.0)
        transformed.append(
            {
                "xyz": [round(float(value), 8) for value in xyz],
                "catalogue_xyz": _catalogue_coordinates(xyz, record.archive_basis),
                "time": int(op.time_reversal),
            }
        )
    transformed.sort(key=lambda item: (item["time"], item["xyz"]))
    return {
        "record_id": record.record_id,
        "operation_count": len(transformed),
        "probe_orbit": transformed,
    }

#### source/pymatgen/symmetry/maggroups.py::MagneticSpaceGroup.symmetry_ops (L304)
    def symmetry_ops(self):
        """
        Retrieve magnetic symmetry operations of the space group.
        :return: List of :class:`pymatgen.core.operations.MagSymmOp`
        """
        ops = [op_data['op'] for op_data in self._data['bns_operators']]

        # add lattice centerings
        centered_ops = []
        lattice_vectors = [l['vector'] for l in self._data['bns_lattice']]

        for vec in lattice_vectors:
            if not (np.array_equal(vec, [1, 0, 0])
                    or np.array_equal(vec, [0, 1, 0])
                    or np.array_equal(vec, [0, 0, 1])):
                for op in ops:
                    new_vec = op.translation_vector + vec
                    new_op = MagSymmOp.from_rotation_and_translation_and_time_reversal(op.rotation_matrix,
                                                                                       translation_vec=new_vec,
                                                                                       time_reversal=op.time_reversal)
                    centered_ops.append(new_op)

        ops = ops + centered_ops

        return ops

def _write_all_magnetic_space_groups_to_file(filename):
    """
    Write all magnetic space groups to a human-readable text file.
    Should contain same information as text files provided by ISO-MAG.
    :param filename:
    :return:
    """
    s = ('Data parsed from raw data from:\n'
         'ISO-MAG, ISOTROPY Software Suite, iso.byu.edu\n'
         'http://stokes.byu.edu/iso/magnetic_data.txt\n'
         'Used with kind permission from Professor Branton Campbell, BYU\n\n')
    all_msgs = []
    for i in range(1, 1652):
        all_msgs.append(MagneticSpaceGroup(i))
    for msg in all_msgs:
        s += '\n{}\n\n--------\n'.format(msg.data_str())
    f = open(filename, 'w')
    f.write(s)
    f.close()

def latexify_spacegroup(spacegroup_symbol):
    r"""
    Generates a latex formatted spacegroup. E.g., P2_1/c is converted to
    P2$_{1}$/c and P-1 is converted to P$\\overline{1}$.

    Args:
        spacegroup_symbol (str): A spacegroup symbol

    Returns:
        A latex formatted spacegroup with proper subscripts and overlines.
    """
    sym = re.sub(r"_(\d+)", r"$_{\1}$", spacegroup_symbol)
    return re.sub(r"-(\d)", r"$\\overline{\1}$", sym)

def unicodeify_spacegroup(spacegroup_symbol):
    r"""
    Generates a unicode formatted spacegroup. E.g., P2$_{1}$/c is converted to
    P2₁/c and P$\\overline{1}$ is converted to P̅1.

    Args:
        spacegroup_symbol (str): A spacegroup symbol as LaTeX

    Returns:
        A unicode spacegroup with proper subscripts and overlines.
    """

    if not spacegroup_symbol:
        return ""

    subscript_unicode_map = {
        0: "₀",
        1: "₁",
        2: "₂",
        3: "₃",
        4: "₄",
        5: "₅",
        6: "₆",
        7: "₇",
        8: "₈",
        9: "₉",
    }

    symbol = latexify_spacegroup(spacegroup_symbol)

    for number, unicode_number in subscript_unicode_map.items():
        symbol = symbol.replace("$_{" + str(number) + "}$", unicode_number)
        symbol = symbol.replace("_" + str(number), unicode_number)

    overline = "\u0305"  # u"\u0304" (macron) is also an option

    symbol = symbol.replace("$\\overline{", overline)
    symbol = symbol.replace("$", "")
    symbol = symbol.replace("{", "")
    symbol = symbol.replace("}", "")

    return symbol
