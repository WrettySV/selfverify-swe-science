from __future__ import annotations

import json
import math
import sys
from pathlib import Path

import numpy as np
import torch


TASK_DIR = Path(__file__).resolve().parent
SOURCE_DIR = TASK_DIR / "source"
FIXTURE = TASK_DIR / "fixtures" / "study_pair.npz"
REPORT = TASK_DIR / "outputs" / "public_reproduction_report.json"
CONSISTENCY_TOLERANCE = 0.12


def write_report(payload: dict[str, object], exit_code: int) -> int:
    REPORT.parent.mkdir(exist_ok=True)
    text = json.dumps(payload, indent=2, sort_keys=True) + "\n"
    REPORT.write_text(text, encoding="utf-8")
    print(text, end="")
    return exit_code


def as_one_hot(labels: np.ndarray) -> torch.Tensor:
    tensor = torch.as_tensor(labels, dtype=torch.int64)
    return torch.nn.functional.one_hot(tensor, num_classes=2).permute(0, 4, 1, 2, 3)


def score_case(metric_cls, prediction: np.ndarray, reference: np.ndarray, spacing: np.ndarray, tolerance: float) -> float:
    metric = metric_cls(
        class_thresholds=[tolerance],
        include_background=False,
        reduction="none",
    )
    result = metric(
        as_one_hot(prediction[None, ...]),
        as_one_hot(reference[None, ...]),
        spacing=tuple(float(value) for value in spacing),
    )
    value = float(result.detach().cpu().reshape(-1)[0])
    if not math.isfinite(value):
        raise ValueError("surface metric returned a non-finite foreground score")
    return value


def main() -> int:
    try:
        sys.path.insert(0, str(SOURCE_DIR))
        from monai.metrics import SurfaceDiceMetric

        with np.load(FIXTURE) as data:
            tolerance = float(data["tolerance"])
            scores = {
                name: score_case(
                    SurfaceDiceMetric,
                    data[f"prediction_{name}"],
                    data[f"reference_{name}"],
                    data[f"{name}_spacing"],
                    tolerance,
                )
                for name in ("fine", "coarse")
            }
    except Exception as exc:  # noqa: BLE001 - report infrastructure failures distinctly.
        return write_report(
            {
                "schema_version": "1.0",
                "status": "runner_failure",
                "error_type": type(exc).__name__,
                "error": str(exc),
            },
            2,
        )

    finite = all(math.isfinite(value) for value in scores.values())
    spread = abs(scores["fine"] - scores["coarse"]) if finite else None
    consistent = finite and spread is not None and spread <= CONSISTENCY_TOLERANCE
    report = {
        "schema_version": "1.0",
        "status": "post_fix_success" if consistent else "pre_fix_expected_failure",
        "scientific_check": "representation_consistency",
        "observations": {
            "finite": finite,
            "study_representations": {
                name: {"surface_score": value} for name, value in scores.items()
            },
            "absolute_representation_spread": spread,
            "consistency_check_passed": consistent,
        },
    }
    return write_report(report, 0 if consistent else 1)


if __name__ == "__main__":
    raise SystemExit(main())

### paper.md
# Scientific context: surface Dice for sampled organ contours

## Citation

Stanislav Nikolov, Sam Blackwell, Alexei Zverovitch, et al., “Deep learning to achieve clinically applicable segmentation of head and neck anatomy for radiotherapy,” arXiv:1809.04430 (2018). The paper is available at <https://arxiv.org/abs/1809.04430> and <https://doi.org/10.48550/arXiv.1809.04430>.

## Why the metric exists

Radiotherapy planning requires contours around organs at risk. An automated contour can be useful in volume terms and still have a clinically important local displacement along its surface. A contour-oriented score therefore complements volume overlap: it asks how much of each boundary lies within an acceptable distance of the other boundary.

The paper calls this family of measurements surface Dice similarity coefficient. Let `S_ref` and `S_pred` be the two boundaries and let `tau` be the allowed physical deviation. A normalized symmetric form is:

```text
surface_Dice(tau) =
  (measure({p in S_ref  : distance(p, S_pred) <= tau})
   + measure({q in S_pred : distance(q, S_ref) <= tau}))
  / (measure(S_ref) + measure(S_pred))
```

The score is between zero and one for ordinary nonempty inputs. The two directed terms matter: a prediction can omit part of a reference boundary or add a boundary outside the reference tolerance.

## Physical coordinates versus array indices

Medical images are sampled on a grid. A voxel index is not a physical length. If the spacing is `(sx, sy, sz)`, moving one index along the three axes represents different physical distances when the spacings differ. A tolerance used to judge a contour is normally expressed in physical units such as millimetres, so spacing must be interpreted consistently throughout the calculation.

The denominator in the formula is a geometric measure of the boundary: a length in 2-D or an area in 3-D. A sampled image therefore needs a documented convention for turning neighboring mask samples into that measure. The physical spacing and the local sampled geometry are part of the input interpretation, especially when the grid is anisotropic. The same convention must be used for the total boundary and for the portion within tolerance. More than one numerical surface construction can satisfy this physical requirement; the scientific contract is the spacing-aware behavior, not a particular implementation detail.

## Inputs in this task

The source works with one-hot, batch-first tensors. A two-dimensional input has shape `[B, C, H, W]`; a three-dimensional input has shape `[B, C, H, W, D]`. The task's public fixture uses a foreground class and supplies the voxel spacing for each complete study representation. The public reproduction is only a smoke test; a correct repair must also work for other masks, spacing tuples, tolerances, and batches.

If a class is absent from both inputs, the metric may represent that class as undefined according to the existing API. If a class is present in only one input, the score should reflect the complete mismatch. These cases are part of the surrounding metric contract and should not be “fixed” by returning a convenient constant.

def compute_mean_error_metrics(y_pred: torch.Tensor, y: torch.Tensor, func: Callable) -> torch.Tensor:
    # reducing in only channel + spatial dimensions (not batch)
