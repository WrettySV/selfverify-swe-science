def load_case() -> dict[str, Any]:
    case = json.loads(FIXTURE_PATH.read_text(encoding="utf-8"))
    required = {
        "case_name",
        "laboratories",
        "protocols",
        "cell_estimates",
        "cell_standard_errors",
        "prior_scales",
        "sampling",
        "planned_contrasts",
    }
    missing = sorted(required - set(case))

        raise ValueError(f"fixture is missing fields: {', '.join(missing)}")

def finite_matrix(np, value: object, shape: tuple[int, int], field: str):
    matrix = np.asarray(value, dtype=float)
    if matrix.shape != shape:
        raise ValueError(f"{field} must have shape {shape}, got {matrix.shape}")
    if not np.all(np.isfinite(matrix)):
        raise ValueError(f"{field} contains non-finite values")
    return matrix


def validate_case(np, case: dict[str, Any]) -> dict[str, Any]:
    laboratories = list(case["laboratories"])
    protocols = list(case["protocols"])
    if len(laboratories) < 3 or len(protocols) < 3:
        raise ValueError("the study requires at least three laboratories and protocols")
    if len(set(laboratories)) != len(laboratories):
        raise ValueError("laboratory names must be unique")
    if len(set(protocols)) != len(protocols):
        raise ValueError("protocol names must be unique")

    shape = (len(laboratories), len(protocols))
    cell_estimates = finite_matrix(np, case["cell_estimates"], shape, "cell_estimates")
    cell_standard_errors = finite_matrix(
        np,
        case["cell_standard_errors"],
        shape,
        "cell_standard_errors",
    )
    if np.any(cell_standard_errors <= 0):
        raise ValueError("cell_standard_errors must be positive")

    prior_scales = dict(case["prior_scales"])
    for field in ("grand_mean", "laboratory", "protocol", "interaction"):
        value = float(prior_scales[field])
        if not np.isfinite(value) or value <= 0:
            raise ValueError(f"prior scale {field} must be positive and finite")
        prior_scales[field] = value

    sampling = dict(case["sampling"])
    for field in ("draws", "tune", "chains", "cores", "seed"):
        sampling[field] = int(sampling[field])
    sampling["target_accept"] = float(sampling["target_accept"])
    if sampling["draws"] < 100 or sampling["tune"] < 0:
        raise ValueError("sampling settings are not sufficient for the public workflow")
    if sampling["chains"] != 1 or sampling["cores"] != 1:
        raise ValueError("the public workflow requires one bounded chain and core")
    if not 0.5 < sampling["target_accept"] < 1.0:
        raise ValueError("target_accept must lie between 0.5 and 1.0")

    laboratory_index = {name: index for index, name in enumerate(laboratories)}
    protocol_index = {name: index for index, name in enumerate(protocols)}
    planned_contrasts = list(case["planned_contrasts"])
    if not planned_contrasts:

        normalized_terms = []
        coefficient_sum = 0.0

            if protocol not in protocol_index:

def flatten_posterior(np, posterior, variable: str, trailing_shape: tuple[int, ...]):
    values = np.asarray(posterior[variable], dtype=float)
    expected_ndim = 2 + len(trailing_shape)
    if values.ndim != expected_ndim:
        raise RuntimeError(
            f"posterior variable {variable} has ndim {values.ndim}, expected {expected_ndim}"
        )
    if trailing_shape and tuple(values.shape[-len(trailing_shape) :]) != trailing_shape:
        raise RuntimeError(
            f"posterior variable {variable} has trailing shape "
            f"{values.shape[-len(trailing_shape):]}, expected {trailing_shape}"
        )
    reshaped = values.reshape((-1,) + trailing_shape) if trailing_shape else values.reshape(-1)

        raise RuntimeError(f"posterior variable {variable} has too few samples")
    if not np.all(np.isfinite(reshaped)):
        raise RuntimeError(f"posterior variable {variable} contains non-finite samples")
    return reshaped

### reproduction.md
# Public reproduction

Run the centered factorial-effects workflow with:

```bash
python reproduce.py
```

The script reads the fixed offline study in
`fixtures/factorial_calibration_study.json`. It builds one Gaussian model with
laboratory, protocol, and laboratory-by-protocol effects, runs bounded
fixed-seed posterior sampling, and evaluates three pre-planned interaction
contrasts.

It writes:

```text
outputs/public_reproduction_report.json
```

The report contains finite sample counts, zero-sum residuals, and posterior
contrast means and standard deviations observed from the candidate model. It
does not contain fixed expected summaries or evaluate an analytic probability
density.

The report uses three statuses:

- `workflow_completed`: the candidate model produced finite posterior
  observations and constraint diagnostics; this is not a correctness claim;
- `pre_fix_expected_failure`: the model ran but did not produce a usable finite
  observation; and
- `runner_failure`: imports, fixture validation, model construction, sampling,
  extraction, shape checks, or report generation failed before a usable
  scientific comparison existed.

The public command is a diagnostic workbench. Authoritative distribution
comparisons and unseen-shape coverage are implemented in the private verifier.
### paper.md
# Centered factorial effects and equivalent Gaussian parameterizations

This offline packet contains the statistical definitions needed to interpret
the supplied calibration workflow. It is based on the version-matched PyMC
distribution documentation, the PyMC3 framework paper, and the Stan User's
Guide discussion of centered vectors.

## 1. Factorial calibration model

Suppose laboratories `l = 1, ..., L` measure the same response under protocols
`p = 1, ..., P`. A two-way Gaussian effects model can be written as

$$
y_{lp} \sim
\mathcal{N}(\mu + a_l + b_p + g_{lp}, s_{lp}),
$$

where `mu` is the grand mean, `a_l` is a laboratory effect, `b_p` is a
protocol effect, `g_lp` is a laboratory-by-protocol interaction, and `s_lp`
is the known standard error of the observed cell summary.

Without identifying constraints, constants can be moved between the grand mean
and the main effects without changing fitted responses. A common convention is

$$
\sum_{l=1}^{L} a_l = 0,
\qquad
\sum_{p=1}^{P} b_p = 0.
$$

The interaction represents only the non-additive part of the response when it
is centered across both axes:

$$
\sum_{l=1}^{L} g_{lp} = 0 \quad \text{for every } p,
\qquad
\sum_{p=1}^{P} g_{lp} = 0 \quad \text{for every } l.
$$

These constraints define the interpretation of the effects. They are not
additional observations.

## 2. Documented centered Gaussian distribution

The source-version PyMC documentation describes `ZeroSumNormal` as a centered
Gaussian distribution for which one or more selected trailing axes sum to zero.
For one constrained axis of length `K`, the covariance is

$$
\operatorname{Cov}(x) =
\sigma^2\left(I - \frac{1}{K}J\right),
$$

where `I` is the identity matrix and every entry of `J` is one. The parameter
`sigma` is the standard deviation of the underlying unconstrained Normal used
by the construction. In this source version it is a positive scalar.

By default the rightmost axis is constrained. The public API can constrain more
than one trailing axis. A matrix constrained over its last two axes therefore
has zero sums along both of those axes.

## 3. Orthonormal reduced coordinates

A zero-sum vector can be represented without redundancy. Let the columns of
`B` form an orthonormal basis for the subspace perpendicular to the all-ones
vector:

$$
B^T B = I,
\qquad
B^T \mathbf{1} = 0.
$$

If

$$
z \sim \mathcal{N}(0, \sigma^2 I),
\qquad
x = Bz,
$$

then `x` has the centered covariance above and lies on the zero-sum support.
Subtracting the mean from an unconstrained Gaussian vector is another way to
construct the same covariance.

For a two-way interaction, bases for the laboratory and protocol centered
subspaces can be combined to represent exactly the matrices whose row and
column sums are zero.

## 4. Reparameterization coherence

A full constrained representation and an orthonormal reduced-coordinate
representation are two coordinate systems for the same statistical model. In a
linear-Gaussian analysis with the same priors, observations, and standard
errors, they must produce the same posterior fitted responses and the same
posterior summaries for any fixed linear contrast, up to numerical and Monte
Carlo error.

This statement does not depend on a particular optimizer or sampler. Different
computational paths may represent the model differently internally, but they do
not define different prior shrinkage or different scientific contrasts.

def quaddist_matrix(cov=None, chol=None, tau=None, lower=True, *args, **kwargs):
    if chol is not None and not lower:
        chol = chol.T

    if len([i for i in [tau, cov, chol] if i is not None]) != 1:
        raise ValueError("Incompatible parameterization. Specify exactly one of tau, cov, or chol.")

    if cov is not None:
        cov = pt.as_tensor_variable(cov)
        if cov.ndim != 2:
            raise ValueError("cov must be two dimensional.")
    elif tau is not None:
        tau = pt.as_tensor_variable(tau)
        if tau.ndim != 2:
            raise ValueError("tau must be two dimensional.")
        # TODO: What's the correct order/approach (in the non-square case)?
        # `pytensor.tensor.nlinalg.tensorinv`?
        cov = matrix_inverse(tau)
    else:
        # TODO: What's the correct order/approach (in the non-square case)?
        chol = pt.as_tensor_variable(chol)
        if chol.ndim != 2:
            raise ValueError("chol must be two dimensional.")

        # tag as lower triangular to enable pytensor rewrites of chol(l.l') -> l
        chol.tag.lower_triangular = True
        cov = chol.dot(chol.T)

    return cov
#### source/pymc/distributions/multivariate.py::LKJCorrRV._random_corr_matrix (L1526)
    def _random_corr_matrix(cls, rng, n, eta, flat_size):
        # original implementation in R see:
        # https://github.com/rmcelreath/rethinking/blob/master/R/distributions.r
        beta = eta - 1.0 + n / 2.0
        r12 = 2.0 * stats.beta.rvs(a=beta, b=beta, size=flat_size, random_state=rng) - 1.0
        P = np.full((flat_size, n, n), np.eye(n))
        P[..., 0, 1] = r12
        P[..., 1, 1] = np.sqrt(1.0 - r12**2)
        for mp1 in range(2, n):
            beta -= 0.5
            y = stats.beta.rvs(a=mp1 / 2.0, b=beta, size=flat_size, random_state=rng)
            z = stats.norm.rvs(loc=0, scale=1, size=(flat_size, mp1), random_state=rng)
            z = z / np.sqrt(np.einsum("ij,ij->i", z, z))[..., np.newaxis]
            P[..., 0:mp1, mp1] = np.sqrt(y[..., np.newaxis]) * z
            P[..., mp1, mp1] = np.sqrt(1.0 - y)
        C = np.einsum("...ji,...jk->...ik", P, P)
        return C

def zerosumnormal_logp(op, values, normal_dist, sigma, support_shape, **kwargs):
    (value,) = values
    shape = value.shape
    n_zerosum_axes = op.ndim_supp

    _deg_free_support_shape = pt.inc_subtensor(shape[-n_zerosum_axes:], -1)
    _full_size = pt.prod(shape)
    _degrees_of_freedom = pt.prod(_deg_free_support_shape)

    zerosums = [
        pt.all(pt.isclose(pt.mean(value, axis=-axis - 1), 0, atol=1e-9))
        for axis in range(n_zerosum_axes)
    ]

    out = pt.sum(
        pm.logp(normal_dist, value) * _degrees_of_freedom / _full_size,
        axis=tuple(np.arange(-n_zerosum_axes, 0)),
    )

    return check_parameters(out, *zerosums, msg="mean(value, axis=n_zerosum_axes) = 0")

    def symbolic_sample_over_posterior(self, node):
        """*Dev* - performs sampling of node applying independent samples from posterior each time.
        Note that it is done symbolically and this node needs :func:`set_size_and_deterministic` call

            sample, random, non_sequences=_known_scan_ignored_inputs(makeiter(random))

        return nodes

#### source/pymc/distributions/continuous.py::get_tau_sigma (L211)
def get_tau_sigma(tau=None, sigma=None):
    r"""
    Find precision and standard deviation. The link between the two
    parameterizations is given by the inverse relationship:

    .. math::
        \tau = \frac{1}{\sigma^2}

    Parameters
    ----------
    tau: array-like, optional
    sigma: array-like, optional

    Results
    -------
    Returns tuple (tau, sigma)

    Notes
    -----
    If neither tau nor sigma is provided, returns (1., 1.)
    """
    if tau is None:
        if sigma is None:
            sigma = 1.0
            tau = 1.0
        else:
            if isinstance(sigma, Variable):
                # Keep tau negative, if sigma was negative, so that it will fail when used
                tau = (sigma**-2.0) * pt.sign(sigma)
            else:
                sigma_ = np.asarray(sigma)
                if np.any(sigma_ <= 0):
                    raise ValueError("sigma must be positive")
                tau = sigma_**-2.0

    else:
        if sigma is not None:
            raise ValueError("Can't pass both tau and sigma")
        else:
            if isinstance(tau, Variable):
                # Keep sigma negative, if tau was negative, so that it will fail when used
                sigma = pt.abs(tau) ** (-0.5) * pt.sign(tau)
            else:
                tau_ = np.asarray(tau)
                if np.any(tau_ <= 0):
                    raise ValueError("tau must be positive")
                sigma = tau_**-0.5

    return floatX(tau), floatX(sigma)

def _handle_sigma_noise_parameters(sigma, noise):
    """Helper function for transition of 'noise' parameter to be named 'sigma'."""

    if (sigma is None and noise is None) or (sigma is not None and noise is not None):
        raise ValueError("'sigma' argument must be specified.")

    if sigma is None:
        warnings.warn(_noise_deprecation_warning, FutureWarning)
        return noise

    return sigma
